** setting up the socket and tunnel for http://gitea.port111.com

*** install (one time)
#+begin_example
# done download mysocket binary
curl https://download.edge.mysocket.io/linux_arm64/mysocketctl -o mysocketctl

# done put the binary someplace on my $path
chmod +x mysocketclt
sudo mv mysocketctl /usr/local/bin/

# done check version, show help
mysocketctl -v
mysocket -h

# done login, show account
mysocketctl login
mysocketctl account show

# done where the secret stuff goes
ls -l ~/.mysocketio_token ~/.mysocket/
#+end_example

** create socket, tunnel (one time per socket)
#+begin_example
# done create a socket
mysocketctl socket create -n "gaulthub"

# done list the socket
mysocketctl socket ls
┌──────────────────────────────────────┬─────────────────────────────────────┬─────────┬──────┬────────────┬──────────┐
│ socket id                            │ dns name                            │ port(s) │ type │ cloud auth │ name     │
├──────────────────────────────────────┼─────────────────────────────────────┼─────────┼──────┼────────────┼──────────┤
│ df5ace52-579e-4203-b031-2d279d946539 │ divine-violet-9569.edge.mysocket.io │ 80, 443 │ http │ false      │ gaulthub │
└──────────────────────────────────────┴─────────────────────────────────────┴─────────┴──────┴────────────┴──────────┘

# done create a tunnel for the socket
mysocketctl tunnel create -s df5ace52-579e-4203-b031-2d279d946539

#+end_example

** Connect the socket (repeated, e.g. on host reboots)
#+begin_example
# DONE connect the socket and log
mysocketctl tunnel connect -s df5ace52-579e-4203-b031-2d279d946539 -t 75b0dd6b-3c65-4a6a-9f74-02f289d1a87b \
  -p 3000 --host gaulthub \
  |& tee logs/2021-11-04-tunnel.log

# DONE list the tunnels for the socket
mysocketctl tunnel ls -s df5ace52-579e-4203-b031-2d279d946539
